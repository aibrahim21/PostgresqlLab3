testdb=# CREATE VIEW student_grade AS 
SELECT student.e_name AS student_name, AVG(grades.grade) AS average_grade , subject.sub_name AS subject_name 
FROM student
JOIN stu_sub ON student.stuid = stu_sub.stu_id
 JOIN subject ON subject.subjectid = stu_sub.sub_id   
JOIN grades ON stu_sub.sub_id = grades.sub_id
GROUP BY student.stuid , subject.sub_name ;
CREATE VIEW
